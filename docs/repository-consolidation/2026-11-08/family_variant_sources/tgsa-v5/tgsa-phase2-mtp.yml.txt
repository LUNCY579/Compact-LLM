name: TGSA Phase 2A trajectory mechanism

on:
  push:
    branches:
      - research/tgsa-phase2a-ci-and-competence
    paths:
      - "Rineya_Trajectory_Guided_Sequential_Attention_Lab_v1/**"
      - ".github/workflows/tgsa-phase2-mtp.yml"
  pull_request:
    paths:
      - "Rineya_Trajectory_Guided_Sequential_Attention_Lab_v1/**"
      - ".github/workflows/tgsa-phase2-mtp.yml"
  workflow_dispatch:

permissions:
  contents: read

concurrency:
  group: tgsa-phase2a-${{ github.ref }}
  cancel-in-progress: false

env:
  LAB: Rineya_Trajectory_Guided_Sequential_Attention_Lab_v1
  CONFIG: Rineya_Trajectory_Guided_Sequential_Attention_Lab_v1/configs/phase2_tgsa_mtp_smoke.json
  CONTRACT: Rineya_Trajectory_Guided_Sequential_Attention_Lab_v1/configs/phase2_program_contract.json
  ROOT: ci_artifacts/tgsa_phase2a_mtp
  PYTHONUNBUFFERED: "1"

jobs:
  unit-and-rehearsal:
    name: Phase 2A semantics, regressions, and development-only rehearsal
    runs-on: ubuntu-latest
    timeout-minutes: 60

    steps:
      - name: Check out full lineage
        uses: actions/checkout@v4
        with:
          fetch-depth: 0

      - name: Set up Python
        uses: actions/setup-python@v5
        with:
          python-version: "3.11"
          cache: pip
          cache-dependency-path: "${{ env.LAB }}/pyproject.toml"

      - name: Install CPU package and test dependencies
        run: |
          python -m pip install --upgrade pip
          python -m pip install --index-url https://download.pytorch.org/whl/cpu "torch>=2.1"
          python -m pip install -e "${LAB}[dev]"

      - name: Verify canonical branch and MTP semantic contract
        run: |
          mkdir -p "${ROOT}"
          python "${LAB}/scripts/verify_phase2_program_contract.py" \
            --contract "${CONTRACT}" \
            --config "${CONFIG}" \
            --output "${ROOT}/program_contract_verification.json"
          python - <<'PY'
          import json
          import os
          import subprocess
          from pathlib import Path

          report = json.loads(
              Path(os.environ["ROOT"] + "/program_contract_verification.json").read_text(encoding="utf-8")
          )
          assert report["passed"] is True, report["errors"]
          assert report["study_kind"] == "mechanism_bridge"
          assert report["mtp_semantics"] == "sequential_teacher_forced_recursive_v1"
          subprocess.run(
              ["git", "merge-base", "--is-ancestor", "8b5737e260b5478427b8bb033cffa394a5f3fc5d", "HEAD"],
              check=True,
          )
          print("Phase 2A semantic identity and v4 result ancestry verified")
          PY

      - name: Run TGSA x MTP and integrity focused tests
        run: |
          python -m pytest \
            "${LAB}/tests/test_mtp.py" \
            "${LAB}/tests/test_mtp_ladder.py" \
            "${LAB}/tests/test_phase2_integrity.py" \
            "${LAB}/tests/test_phase2_program_contract.py" \
            -q --tb=short --durations=15

      - name: Run complete TGSA regression suite
        run: |
          python -m pytest "${LAB}/tests" -q --tb=short --durations=15

      - name: Run the eight-cell non-evidentiary Phase 2A rehearsal
        run: |
          rm -rf "${ROOT}/dataset" "${ROOT}/study"
          python "${LAB}/scripts/phase2_tgsa_mtp_rehearsal.py" \
            --config "${CONFIG}" \
            --dataset-dir "${ROOT}/dataset" \
            --output-dir "${ROOT}/study" \
            --repo-root "${GITHUB_WORKSPACE}"

      - name: Verify Phase-2 integrity and anti-laundering contracts
        run: |
          python "${LAB}/scripts/verify_phase2_integrity.py" \
            --config "${CONFIG}" \
            --dataset-dir "${ROOT}/dataset" \
            --output-dir "${ROOT}/study"

      - name: Validate Phase-2 artifacts against JSON schemas
        run: |
          python - <<'PY'
          import json
          import os
          from pathlib import Path

          import jsonschema

          lab = Path(os.environ["LAB"])
          root = Path(os.environ["ROOT"])
          study = root / "study"
          run_schema = json.loads(
              (lab / "evidence/phase2_mtp_run.schema.json").read_text(encoding="utf-8")
          )
          report_schema = json.loads(
              (lab / "evidence/phase2_mtp_report.schema.json").read_text(encoding="utf-8")
          )
          integrity_schema = json.loads(
              (lab / "evidence/phase2_integrity.schema.json").read_text(encoding="utf-8")
          )
          program_schema = json.loads(
              (lab / "evidence/phase2_program_contract.schema.json").read_text(encoding="utf-8")
          )
          resolved_run_schema_id = (
              "https://rineya.invalid/schemas/phase2_mtp_run.schema.json"
          )
          resolver = jsonschema.RefResolver.from_schema(
              report_schema,
              store={
                  "phase2_mtp_run.schema.json": run_schema,
                  resolved_run_schema_id: run_schema,
                  run_schema["$id"]: run_schema,
              },
          )
          report = json.loads((study / "phase2_report.json").read_text(encoding="utf-8"))
          jsonschema.validate(report, report_schema, resolver=resolver)
          integrity = json.loads((study / "phase2_integrity.json").read_text(encoding="utf-8"))
          jsonschema.validate(integrity, integrity_schema)
          program = json.loads((root / "program_contract_verification.json").read_text(encoding="utf-8"))
          jsonschema.validate(program, program_schema)
          manifests = sorted(study.glob("runs/*/seed_*/run_manifest.json"))
          assert manifests, "no run manifests were produced"
          for manifest_path in manifests:
              jsonschema.validate(
                  json.loads(manifest_path.read_text(encoding="utf-8")),
                  run_schema,
              )
          print(
              f"validated one programme contract, one report, one integrity manifest, "
              f"and {len(manifests)} runs"
          )
          PY

      - name: Verify competence, initialization, gradient, leakage, and dose contracts
        run: |
          python - <<'PY'
          import json
          import os
          from pathlib import Path

          root = Path(os.environ["ROOT"])
          report = json.loads(
              (root / "study/phase2_report.json").read_text(encoding="utf-8")
          )
          integrity = json.loads(
              (root / "study/phase2_integrity.json").read_text(encoding="utf-8")
          )
          verification = json.loads(
              (root / "study/phase2_integrity_verification.json").read_text(encoding="utf-8")
          )
          program = json.loads(
              (root / "program_contract_verification.json").read_text(encoding="utf-8")
          )
          assert program["passed"] is True
          assert program["study_kind"] == "mechanism_bridge"
          assert program["mtp_semantics"] == "sequential_teacher_forced_recursive_v1"
          assert verification["passed"] is True, verification["errors"]
          assert report["status"] == "complete", report
          assert report["confirmatory"] is False
          assert report["development_only"] is True
          assert report["mechanism_interpretation_allowed"] is False
          assert report["decision"] == "smoke_only"
          assert report["competence_precondition"]["required"] is False
          assert report["external_phase1_competence"]["required"] is False
          assert len(report["runs"]) == 8
          assert report["analysis"]["decision"] == "descriptive_only"
          assert report["training_dose"]["sample_presentations"] == 12
          assert report["training_dose"]["unique_train_examples"] == 48
          assert report["training_dose"]["effective_epochs"] == 0.25
          assert len(report["regime_fingerprint"]) == 64
          assert len(report["device_fingerprint"]) == 64
          assert len(report["study_fingerprint"]) == 64

          assert integrity["development_only"] is True
          assert integrity["mechanism_interpretation_allowed"] is False
          assert integrity["contract"]["split_generation_order"] == ["test", "dev", "train"]

          for audit in report["initialization_audits"]:
              assert audit["all_mtp_heads_identical"] is True
              assert audit["backbone_identical_within_tgsa_arm"] is True

          fingerprints = set()
          for run in report["runs"]:
              gradient = run["gradient_audit"]
              assert run["development_only"] is True
              assert run["mechanism_interpretation_allowed"] is False
              assert run["training_dose"] == report["training_dose"]
              assert run["integrity"]["regime_fingerprint"] == report["regime_fingerprint"]
              assert run["integrity"]["device_fingerprint"] == report["device_fingerprint"]
              assert run["metric_semantics"]["wall_clock_speed_claim"] is False
              fingerprints.add(run["integrity"]["cell_fingerprint"])
              assert gradient["mtp_head_gradient_norm"] > 0
              if run["cell"]["mtp_mode"] == "trajectory":
                  assert gradient["trajectory_control_requires_grad"] is True
                  assert gradient["trajectory_control_gradient_norm"] > 0
                  assert run["evaluation"]["trajectory_control_rms"] > 0
              else:
                  assert run["evaluation"]["trajectory_control_rms"] == 0
              scope = run["evaluation"]["teacher_forced_future_prefix"]["claim_scope"]
              assert scope.endswith("not throughput")
          assert len(fingerprints) == len(report["runs"])

          forbidden = list((root / "study").rglob("predictions_test*.jsonl"))
          assert not forbidden, forbidden
          assert (root / "dataset/test.jsonl").is_file()
          print("TGSA Phase 2A integrity and claim contracts verified")
          PY

      - name: Upload rehearsal evidence
        if: always()
        uses: actions/upload-artifact@v4
        with:
          name: tgsa-phase2a-mtp-${{ github.run_id }}
          path: ci_artifacts/tgsa_phase2a_mtp
          if-no-files-found: warn
          retention-days: 14
