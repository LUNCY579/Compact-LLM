def pointer(path,*,directory=False,parent=None):
    """Reject every symlink/reparse ancestor before resolving selected evidence."""
    path=Path(path);require(path.is_absolute() and ".." not in path.parts,"absolute nonescaping evidence pointer")
    for part in (path,*path.parents):
        details=part.lstat()
        require(not stat.S_ISLNK(details.st_mode) and not getattr(details,"st_file_attributes",0)&0x400,
            "symlink/reparse evidence pointer forbidden")
    require(path.resolve()==path and (path.is_dir() if directory else path.is_file()),"exact regular evidence pointer")
    if parent is not None:require(path.parent==Path(parent),"selected evidence belongs to exact owned leaf")
    return path


def bytes_info(path,maximum=32*1024**2):
    path=pointer(path);before=path.stat();require(0<=before.st_size<=maximum,"bounded immutable selected file")
    value=hashlib.sha256();total=0
    with path.open("rb") as stream:
        for raw in iter(lambda:stream.read(1024**2),b""):total+=len(raw);value.update(raw)
    after=path.stat()
    require(total==before.st_size==after.st_size and (before.st_mtime_ns,before.st_ino)==(after.st_mtime_ns,after.st_ino),
        "selected file changed while hashing")
    return {"bytes":total,"sha256":value.hexdigest()}
